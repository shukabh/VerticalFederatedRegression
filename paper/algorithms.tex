\begin{algorithm}[p]
\caption{\textsc{Setup} (Phase~0)}
\label{alg:setup}
\small
\begin{algorithmic}[1]
\State \Bdo agree on the parameter vector $\mathcal P$ (Table~\ref{tab:params}): the scope $\Phi$ (public attributes only), the caps $n_R^{\max}$ and $\bar n_O$, the budget $(\varepsilon_{\mathrm{run}},\delta_{\mathrm{run}})$, the ledger parameters $(\varepsilon_L,\delta_L,\zeta_L,\xi_L,\delta')$, $f_1$ (and $f_y$), $\zeta$, $\kappa_{\min}$ (or a fixed $\kappa$), $\delta_B$, $t_\eta$, $\vartheta^\star$, $\alpha_{\mathrm{CI}},\alpha_L,\alpha_U$, $B_\beta^{\mathrm{pub}}$, $\bar\sigma^2$, $\kappa_\Psi$, $\eta_B$, $B$, the PSI parameters, $N$, the chain, $\Omega_U$ and $\Omega_u$
\State \Odo $D_\Phi\gets\{j:\Phi(j)\}$; pad with dummy rows (fresh random identifiers, $\bx_O=\mathbf 0$, $y=0$) to $\bar n_O$; draw a fresh secret permutation $\tau$ of $[\bar n_O]$
\State \Bdo optional: screen the features on public reference data that carries both schemas and drop nearly collinear ones \Comment{no budget}
\State \Rdo and \Odo, each on its own columns only: $\textsc{Preprocess}$ with constants from public reference data or domain knowledge
\State \Bdo $(B_R,B_O,B_y)\gets\textsc{ChooseBounds}$ on the preprocessed reference data, or from domain knowledge \Comment{$B_R$ covers $\Rp$'s intercept}
\State $(\Delta,\Delta_{\neg yy})\gets\textsc{DeltaRep}(B_R,B_O,B_y)$;\quad $\Delta_G\gets\textsc{DeltaRep}(B_R,B_O,0)$;\quad $\Delta_y\gets B_y^2$
\State $\textit{mode}\gets\textsf{joint}$ if $\Delta-\Delta_{\neg yy}\le10^{-9}\Delta$, otherwise $\textsf{separate}$ \Comment{release $\by^\top\by$ jointly only when it is free}
\State \Odo $(\kappa,\Omega_V,\Omega_f,\{\sigma_{c,s},\rho_s\}_s,\mathit{cost})\gets\textsc{Calibrate}(\mathcal P)$; refuse if it refuses \Comment{$\Rp$ repeats it and checks}
\State \Odo atomically: \textbf{if} the ledger of the base register admits $\mathit{cost}=(\rho_{\mathrm{run}},\varepsilon'_{\mathrm{run}},\gamma_L)$ (Section~\ref{sec:privacy:ledger}) \textbf{then} charge it in full, for every planned release, \textbf{else} refuse \Comment{no refund}
\State \Bdo threshold key generation with a common random $a$: $s=s_R+s_O$, $\pk=(p_0,a)$ with $p_0=-as+e$, and rotation keys for $\{2^i\}$
\Statex \hspace{\algorithmicindent}($\Rp$ sends $p_{0,R}$ and its rotation-key shares; $\Op$ sends only $p_{0,O}$ and keeps the joint rotation keys; no relinearization or conjugation key)
\State \Odo fresh OPRF key $k$; PSI preprocessing on the items $\{F_k(\mathrm{id}_j)\}$ with labels $\tau(j)$ encrypted under keys derived from $F'_k(\mathrm{id}_j)$; bin sizes and partition counts public; log the run against the OPRF quota of (researcher, project)
\end{algorithmic}
\end{algorithm}

% --------------------------------------------------------------
\begin{algorithm}[p]
\caption{\textsc{Link}: labeled PSI with an OPRF (Phase~1)}
\label{alg:link}
\small
\begin{algorithmic}[1]
\State \Rdo pad the cohort identifiers with random dummies to exactly $n_R^{\max}$
\State \Rdo run the OPRF with $\Op$ on each padded identifier $\to F_k(\mathrm{id}),F'_k(\mathrm{id})$ \Comment{$\Op$ evaluates exactly $n_R^{\max}$ calls}
\State \Rdo cuckoo-hash the items $F_k(\mathrm{id})$ into the bins; send BFV encryptions of their windowed powers under $\Rp$'s own BFV key
\State \Odo evaluate the membership and label polynomials of every bin and partition; mask; $\textsc{SanitizeBFV}$; reply \Comment{flooding: $\kappa_\psi$}
\State \Rdo decrypt $\to$ the match set $I$ and positions $\pi(i)=\tau(j)$ for $i\in I$ (labels opened with keys from $F'_k(\mathrm{id}_i)$)
\State \Rdo $\bA\gets\sum_{i\in I}\bxd_i\bxd_i^\top$, exact and local \Comment{never transmitted, never noised}
\end{algorithmic}
\end{algorithm}

% --------------------------------------------------------------
\begin{algorithm}[p]
\caption{\textsc{Upload} (Phase~2)}
\label{alg:upload}
\small
\begin{algorithmic}[1]
\For{$t=1,\dots,\bar n_O$}
  \State \Rdo $b_t\gets\mathbb{1}[t\in\pi(I)]$;\quad $\bxd_t\gets\bigl(1,\ \mathrm{clip}(\bx_{R,i};\sqrt{B_R^2-1})\bigr)$ if $t=\pi(i)$, otherwise $\mathbf 0$
\EndFor
\State $m\gets\lceil\bar n_O/N_s\rceil$ \Comment{chunks of at most $N_s=N/2$ slots}
\For{each vector $\mathbf v\in\{\bb,\bxd_{\cdot0},\dots,\bxd_{\cdot d_R}\}$ and each chunk $h\le m$}
  \State \Rdo draw a seed $\varsigma$; $a\gets\mathrm{PRG}(\varsigma)$; $c_0\gets-a\,s_R+e+\Ecd_{\Omega_U}(\mathbf v[h])$, with $e$ from the RLWE error distribution; send $(c_0,\varsigma)$
  \State \Odo $a\gets\mathrm{PRG}(\varsigma)$;\ $c_0\gets c_0-a\,s_O$ \Comment{$(c_0,a)$ now encrypts $\mathbf v[h]$ under $s=s_R+s_O$}
\EndFor
\Statex $\Rp$ encodes at the parameter $\Omega_U$ (Section~\ref{sec:privacy:calib})
\end{algorithmic}
\end{algorithm}

% --------------------------------------------------------------
\begin{algorithm}[p]
\caption{\textsc{Aggregate} and pack ($\Op$, Phase~3)}
\label{alg:aggregate}
\small
\begin{algorithmic}[1]
\Require entry map $e\mapsto(U_e,V_e)$, with $\Rp$'s ciphertext vector $U_e$ and $\Op$'s plaintext vector $V_e$ in $\tau$-order:
\Statex \quad $\bB_{jk}$: $(\Enc{\bb},\ \bx_{O,j}\odot\bx_{O,k})$;\quad $\bC_{jk}$: $(\Enc{\bxd_j},\ \bx_{O,k})$;\quad $\bc_{R,j}$: $(\Enc{\bxd_j},\ \by)$;
\Statex \quad $\bc_{O,k}$: $(\Enc{\bb},\ \bx_{O,k}\odot\by)$;\quad $\by^\top\by$: $(\Enc{\bb},\ \by\odot\by)$
\For{each $e\in\mathcal E$}
  \State $T_e\gets\textsc{EvalSum}\bigl(\textsc{Rescale}\bigl(\sum_{h\le m}U_e[h]\odot\Ecd_{\Omega_V}(V_e[h])\bigr)\bigr)$ \Comment{sum the chunks first; one \textsc{EvalSum} per entry}
  \State $P_e\gets\textsc{Rescale}\bigl(T_e\odot\Ecd_{\Omega_u}(\mathbf u_{\mathrm{slot}(e)})\bigr)$ \Comment{$\mathbf u_k$: one-hot slot $k$; entry $e$ sits alone in its slot}
\EndFor
\State $M_1\gets\sum_{e\in\mathcal E_G}P_e$;\quad $M_2\gets\sum_{e\in\mathcal E_2}P_e$;\quad $M_y\gets P_{\by^\top\by}$ if $\textit{mode}=\textsf{separate}$ \Comment{depth 2; $M_1$ depends only on $\triu\bB$ and $\bC$}
\Statex Scales: $\Rp$'s vectors at $\Omega_U$, the masks at $\Omega_u$ (parameters), $\Op$'s data at $\Omega_V$ (from \textsc{Calibrate}); the final scale $\Omega_f=\Omega_U\Omega_V\Omega_u/(q_1q_2)$ is tracked exactly, and $\Rp$ decodes with it
\end{algorithmic}
\end{algorithm}

% --------------------------------------------------------------
\begin{algorithm}[p]
\caption{\textsc{Release}$(M,\sigma_c)$: one DP release (Phase~4)}
\label{alg:release}
\small
\begin{algorithmic}[1]
\State \Odo sample fresh $u$ from the key distribution and $e_0,e_1$ from the error distribution
\State \Odo $c_0'\gets c_0+p_0u+e_0$;\quad $c_1'\gets c_1+au+e_1$ \Comment{re-randomize $M=(c_0,c_1)$ under the joint $\pk$}
\State \Odo $\eta\gets\textsc{SampleDGBudget}(\sigma_c^2,N)$, fresh; \textbf{refuse} the release if it fails \Comment{exact on $\Z^N$; fixed work; fails w.p.\ $\le\xi_T$}
\State \Odo $h_O\gets c_0'+c_1's_O+\eta\bmod q_0$
\State \Odo send $(c_1',h_O)$; erase $c_0',u,e_0,e_1,\eta$ \Comment{$c_0'$ is never sent; the ledger was charged at admission}
\State \Rdo $v\gets h_O+c_1's_R\bmod q_0$;\quad \Return $\Dcd(v)$ restricted to the slots of this release \Comment{exact final scale; $v$ never leaves $\Rp$}
\State \Rdo before publishing any decoded value, round it to a fixed precision relative to $\sigma_s$; never publish $v$ \Comment{Remark~\ref{rem:publish}; no budget}
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[p]
\caption{\textsc{Solve} and \textsc{Infer} ($\Rp$, Phase~5)}
\label{alg:solve}
\small
\begin{algorithmic}[1]
\Require $\bA$, $n$, $p$, slot-domain s.d.\ $\sigma_1,\sigma_2$ (and $\sigma_y$); from $\mathcal P$: $\vartheta^\star$, $\alpha=\alpha_{\mathrm{CI}}+\alpha_L+\alpha_U$, $B_\beta^{\mathrm{pub}}$, the variance bound $\bar\sigma^2$, $\kappa_\Psi$, bootstrap truncation $\eta_B$ and size $B$
\State $(\hat\bB_1,\hat\bC_1)\gets\textsc{Release}(M_1,\sigma_{c,1})$;\quad $\hat\bG_1\gets\begin{psmallmatrix}\bA&\hat\bC_1\\ \hat\bC_1^\top&\hat\bB_1\end{psmallmatrix}$
\State $r_1\gets\sigma_1\bigl(2\sqrt p+2\sqrt{\ln(1/\alpha_L)}\bigr)$, or $r_1\gets\sigma_1q_{1-\alpha_L}$ (exact quantile) \Comment{$\Pr[\lambda_{\max}(\hat\bG_1-\bG)>r_1]\le\alpha_L$ (one-sided)}
\State $\hat L\gets\max\{0,\ \lambda_{\min}(\hat\bG_1)-r_1\}$;\quad $g^\star\gets2\vartheta^\star\sigma_2\sqrt p$
\State $\gamma\gets g^\star+r_1$;\quad $\lambda_I\gets\max\{0,\ \gamma-\lambda_{\min}(\hat\bG_1)\}$
\State $\lambda_O\gets\max\{0,\ \gamma-\lambda_{\min}(\bS_\gamma)\}$ with $\bS_\gamma=\hat\bB_1-\hat\bC_1^\top(\bA-\gamma\bI)^{-1}\hat\bC_1$ if $\lambda_{\min}(\bA)>\gamma$, otherwise $\lambda_O\gets\infty$ \Comment{closed form: the smallest $\lambda\ge0$ with $\lambda_{\min}(\hat\bG_1+\lambda\PiO)\ge\gamma$}
\State $(\lambda,\Psi)\gets(\lambda_O,\PiO)$ if $\lambda_O\le\kappa_\Psi\lambda_I$, otherwise $(\lambda_I,\bI)$ \Comment{uses stage 1 and $\bA$ only}
\State $(\hat\bB_2,\hat\bC_2,\hat\bc_R,\hat\bc_O,\widehat{yy})\gets\textsc{Release}(M_2,\sigma_{c,2})$; if separate, $\widehat{yy}\gets\textsc{Release}(M_y,\sigma_{c,y})$
\State assemble $\hat\bG_2$, $\hat\bc_2$;\quad $\hat\vartheta\gets\lambda_{\min}(\hat\bG_2+\lambda\Psi)/(2\sigma_2\sqrt p)$;\quad \textbf{if} $\hat\vartheta<1$ \textbf{then return} \textsf{refused} \Comment{budget is spent}
\State $\tilde\bP\gets(\hat\bG_2+\lambda\Psi)^{-1}$;\quad $\tilde\btheta\gets\tilde\bP\hat\bc_2$ \Comment{the bias correction of Appendix~\ref{sec:bias} is an option}
\State $\widetilde{\mathrm{RSS}}\gets\widehat{yy}-\hat\bc_2^\top\tilde\btheta-\lambda\,\tilde\btheta^\top\Psi\tilde\btheta$;\quad $\sigma_{yy}\gets\sigma_2$ (joint) or $\sigma_y$ (separate)
\For{$b=1,\dots,B$} \Comment{parametric bootstrap of the mechanism: post-processing}
  \State draw $\mathbf E^*$ (symmetric, s.d.\ $\sigma_2$ on the $\Op$-dependent entries, zero on $\bA$), $\mathbf f^*\sim\mathcal N(0,\sigma_2^2\bI)$, $g^*\sim\mathcal N(0,\sigma_{yy}^2)$
  \State \textbf{keep} the draw only if $\|\tilde\bP^{1/2}\mathbf E^*\tilde\bP^{1/2}\|_{\mathrm{op}}\le\eta_B$ \Comment{truncation: otherwise the moments do not exist}
  \State $\btheta^*_b\gets(\hat\bG_2+\mathbf E^*+\lambda\Psi)^{-1}(\hat\bc_2+\mathbf f^*)$;\quad $\mathrm{RSS}^*_b\gets$ the RSS formula at $(\hat\bG_2+\mathbf E^*,\hat\bc_2+\mathbf f^*,\widehat{yy}+g^*)$
\EndFor
\State $\bV_{\mathrm{mech}}\gets\widehat{\mathrm{Cov}}_b(\btheta^*_b)$;\quad $\mathrm{Corr}\gets\overline{\mathrm{RSS}^*}-\widetilde{\mathrm{RSS}}$;\quad $\tilde s^2\gets\max\{\widetilde{\mathrm{RSS}}-\mathrm{Corr},0\}/(n-p)$
\State \textbf{CI for the protocol estimand} $\btheta_\lambda=(\bG+\lambda\Psi)^{-1}\bc$:\quad $\tilde\theta_j\pm z_{1-\alpha_{\mathrm{CI}}/2}\sqrt{(\bV_{\mathrm{mech}})_{jj}}$
\State $\bV\gets\bV_{\mathrm{mech}}+\bar\sigma^2\,\tilde\bP\hat\bG_2\tilde\bP$ \Comment{sampling variance from the public bound $\bar\sigma^2\ge\max_i\operatorname{Var}(\tilde y_i\mid\bX)$}
\State $\bV^{\mathrm{wm}}\gets\bV_{\mathrm{mech}}+\tilde s^2\,\tilde\bP\hat\bG_2\tilde\bP$ \Comment{sampling variance under the homoskedastic working model}
\State $U_{yy}\gets\min\bigl\{\widehat{yy}+z_{1-\alpha_U/2}\,\sigma_{yy}+B_y^2\sqrt{n\ln(2/\alpha_U)/2},\ nB_y^2\bigr\}$;\quad $\hat B_\beta\gets\min\{B_\beta^{\mathrm{pub}},\sqrt{U_{yy}/\hat L}\}$ \Comment{$\infty$ if $\hat L=0$; $U_{yy}$ valid at level $\alpha_U$}
\State $\bar b_j\gets\lambda\,\|\Psi\tilde\bP\mathbf e_j\|_2\,\hat B_\beta$ \Comment{worst-case ridge bias of coordinate $j$ (first order)}
\State \textbf{CI for} $\beta_j$ (bias-aware): $\tilde\theta_j\pm\mathrm{cv}_{\alpha_{\mathrm{CI}}}\bigl(\bar b_j/\sqrt{V_{jj}}\bigr)\sqrt{V_{jj}}$, where $\mathrm{cv}_\alpha(t)$ is the $(1-\alpha)$ quantile of $|Z+t|$, $Z\sim\mathcal N(0,1)$ \Comment{valid whether or not clipping binds}
\State \textbf{working-model CI for} $\beta_j$: the same with $\bV^{\mathrm{wm}}$ in place of $\bV$ \Comment{valid only under the working model}
\State \Return $\tilde\btheta$, the three intervals, each labelled, $(\lambda,\Psi)$, the shrinkage $-\lambda\tilde\bP\Psi\tilde\btheta$, $\hat\vartheta$, $\sigma_2\tr(\tilde\bP)$, and the $(\varepsilon,\delta)$ spent
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[p]
\caption{Subroutines}
\label{alg:sub}
\small
\begin{algorithmic}[1]
\Function{DeltaRep}{$B_R,B_O,B_y$} \Comment{replace-one of $(\bx_O,y)$, $\bxd$ fixed; $r,o,w=B_R^2,B_O^2,B_y^2$; Proposition~\ref{prop:sens}}
  \State \textbf{if} $|r-w|\le2o$: $\Delta^2=(2o+r+w)^2/2+2rw$;\quad \textbf{else if} $r-w\ge2o$: $\Delta^2=4r(o+w)$
  \State \textbf{else}: exact candidate enumeration (three cubics), rounded up as a safeguard. Return $\Delta$, and $\Delta_{\neg yy}$ from the same maximization without the $(y^2-y'^2)^2$ term
  \State \Comment{exact suprema for $d_O\ge2$, valid but conservative for $d_O=1$; $\Delta_G=(2o+r)/\sqrt2$ if $r\le2o$, otherwise $2\sqrt{ro}$}
\EndFunction
\Statex
\Function{Calibrate}{$\mathcal P$} \Comment{public values only; refuses when a step has no solution}
  \State \textit{mode}, $n_{\mathrm{rel}}\in\{2,3\}$ and $\{\Delta_s\}$ as in \textsc{Setup};\quad \textbf{refuse if} $\log_2(QP)>L_{\max}(N,\lambda_{\mathrm{sec}})$
  \State $\gamma(k):=n_{\mathrm{rel}}\bigl(\delta_B+2^{-k}(1+2^{-80})+2^{-97}+\xi_T\bigr)$;\quad $\varepsilon'(k):=\min\{\varepsilon:\delta(\pi4^{-k},\varepsilon)\le\delta'\}$ \Comment{per run; per release}
  \State \textbf{unless} $\mathcal P$ fixes $\kappa$: $\kappa\gets\max\{\kappa_{\min},\kappa_{\mathrm{run}},\kappa_L\}$ \Comment{\eqref{eq:kappa}; refuse if none exists}
  \Statex \hspace{\algorithmicindent}$\kappa_{\mathrm{run}}=\min\{k:(1+e^{\varepsilon_{\mathrm{run}}})\gamma(k)\le\zeta\delta_{\mathrm{run}}\}$;\quad $\kappa_L=\min\{k:n_{\mathrm{rel}}\varepsilon'(k)\le(\tilde\rho_{\mathrm{run}}/\rho_L)\,E'_L\}$,
  \Statex \hspace{\algorithmicindent}with $\rho_L$, $E'_L=\xi_L\varepsilon_L/2$ and $\Gamma_L$ of Section~\ref{sec:privacy:ledger}; \textbf{refuse if} $n_{\mathrm{rel}}(\delta_B+\delta'+\xi_T)>(\tilde\rho_{\mathrm{run}}/\rho_L)\,\Gamma_L$
  \State $\gamma\gets\gamma(\kappa)$;\quad $\delta_{\mathrm{DP}}\gets\delta_{\mathrm{run}}-(1+e^{\varepsilon_{\mathrm{run}}})\gamma$;\quad \textbf{refuse if} $\delta_{\mathrm{DP}}\le0$
  \State $\rho_{\mathrm{run}}\gets\max\{\rho:\ \inf_{a>1}\frac{e^{(a-1)(a\rho-\varepsilon_{\mathrm{run}})}}{a-1}\bigl(1-\tfrac1a\bigr)^{a}\le\delta_{\mathrm{DP}}\}$ \Comment{tight zCDP conversion}
  \State $\rho_1\gets f_1\rho_{\mathrm{run}}$;\quad $\rho_y\gets f_y\rho_{\mathrm{run}}$ (0 in joint mode);\quad $\rho_2\gets\rho_{\mathrm{run}}-\rho_1-\rho_y$
  \For{$\Omega_f$ in increasing admissible scales} \Comment{$\Omega_V=\Omega_fq_1q_2/(\Omega_U\Omega_u)$, exactly representable}
    \State for each release $s$: $\widehat\Delta_{\mathrm{enc},s}\gets\sqrt{2/N}\,\Omega_f\Delta_s+\Delta_{x,s}+\sqrt N$;\quad $\sigma_{c,s}^2\gets\lceil\widehat\Delta_{\mathrm{enc},s}^2/(2\rho_s)\rceil$
    \State \textbf{if} $\sigma_{c,s}\ge\max\{2^{\kappa}B_{\mathrm{tot},s}(\Omega_f)/\sqrt{2\pi},\,2^{42}\}$ for every $s$ \textbf{then break} \Comment{worst-case $B_{\mathrm{tot},s}$}
  \EndFor \Comment{refuse if no scale floods}
  \State $\sigma_s\gets\sigma_{c,s}\sqrt{N/2}/\Omega_f$;\quad \textbf{refuse unless} $q_0>2\bigl(\Omega_fS_{\max}+2^{\kappa}B_{\mathrm{tot},\max}+t_\eta\sigma_{c,\max}\bigr)$
  \Statex \hspace{\algorithmicindent}with $S_{\max}=\bar n_O\max(r,o,w,\sqrt{ro},\sqrt{rw},\sqrt{ow})$ \Comment{no wraparound}
  \State $T_s\gets$ the work budget of \textsc{SampleDGBudget}$(\sigma_{c,s}^2,N)$ at $\xi_T$ \Comment{public; $\Rp$ checks it}
  \State \Return $\kappa$, $\Omega_V$, $\Omega_f$, $\{\sigma_{c,s}^2,\sigma_s,\rho_s,T_s\}_s$ and the run's ledger cost $(\rho_{\mathrm{run}},\ n_{\mathrm{rel}}\varepsilon'(\kappa),\ n_{\mathrm{rel}}(\delta_B+\delta'+\xi_T))$
\EndFunction
\Statex
\Function{SampleDGBudget}{$\sigma^2,n$} $n$ exact samples of $\mathcal N_{\Z}(0,\sigma^2)$, integer $\sigma^2$, with a fixed work budget
  \State $t\gets2^b\approx\sigma$;\quad $c\gets\lceil2^{128}\sigma^2/(2t^2)\rceil/2^{128}$;\quad $(T,J_v,k_{\max})$ public functions of $(\sigma^2,n,\xi_T)$
  \For{$j=1,\dots,T$, every one with the same fixed-width operations} \Comment{128-bit coins from a CSPRNG}
    \State sign $s$;\ $u\gets\mathrm U[0,t)$;\ $v\gets\#\{1\le i\le J_v:U<e^{-i}\}$;\ $y_j\gets u+tv$ if $s=0$, else $-(u+tv+1)$
    \State $g\gets y_j^2/(2\sigma^2)-v+c\ (\ge0)$;\quad $a_j\gets\mathrm{Bern}(e^{-\lfloor g\rfloor})\wedge\mathrm{Bern}(e^{-(g-\lfloor g\rfloor)})$ \Comment{the latter with exactly $k_{\max}$ coins}
  \EndFor
  \State \textbf{fail} if a trial is undecided or $\sum_ja_j<n$;\quad \Return the first $n$ values $y_j$ with $a_j=1$ \Comment{$\Pr[\textbf{fail}]\le\xi_T$}
\EndFunction
\Statex
\Function{SanitizeBFV}{$\mathit{ct}$} \Comment{statistical flooding; BFV decryption stays exact}
  \State $\mathit{ct}\gets\mathsf{ModSwitch}_{q'}(\mathit{ct})\boxplus\mathsf{ModSwitch}_{q'}\bigl(\mathsf{BFV.Enc}_{\pk_B}(\mathbf 0)\bigr)$;\quad $\mathbf e\gets\textsc{SampleDGBudget}(\sigma_\psi^2,N_B)$ \Comment{switch both, then add; abort on failure}
  \State \Return $\mathit{ct}\boxplus(\mathbf e,0)$ \Comment{$\sigma_\psi\ge\max\{2^{\kappa_\psi}B_\psi/\sqrt{2\pi},2^{42}\}$, $q'/(2t)>t_\eta\sigma_\psi+B_\psi+t$ (Section~\ref{sec:input})}
\EndFunction
\Statex
\Function{Preprocess}{own columns, $D_{\mathrm{ref}}$} \Comment{each party separately; all constants public}
  \State standardize each column with constants from $D_{\mathrm{ref}}$ or domain knowledge ($\Op$ also standardizes $y$)
  \State optional tail transform per column: winsorize at $|z|\le c$, rank-Gaussian from the public quantiles, or log
  \State optional block normalization: scale the party's block of every row to unit norm ($\Rp$ without the intercept)
\EndFunction
\Statex
\Function{ChooseBounds}{$D_{\mathrm{ref}}$} the $99.5\%$ quantiles of $\|\bxd\|$, $\|\bx_O\|$ and $|y|$ in the preprocessed reference data
\Statex \hspace{\algorithmicindent}(the evaluation setting; quantile levels that minimize the first-order MSE plus the squared clipping bias are future work)
\EndFunction
\end{algorithmic}
\end{algorithm}
