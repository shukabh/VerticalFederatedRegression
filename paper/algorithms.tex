\begin{algorithm}[p]
\caption{\textsc{Setup} (Phase~0)}
\label{alg:setup}
\small
\begin{algorithmic}[1]
\State \Bdo agree on the scope $\Phi$ (public attributes only), the caps $n_R^{\max}$ and $\bar n_O$, the budget $(\varepsilon_{\mathrm{run}},\delta_{\mathrm{run}})$, the stage-1 share $f_1$ (and $f_y$), the validity target $\vartheta^\star$, the error level $\alpha$, the public bound $B_\beta^{\mathrm{pub}}$ and the ratio $\kappa_\Psi$
\State \Odo $D_\Phi\gets\{j:\Phi(j)\}$; pad with dummy rows (fresh random identifiers, $\bx_O=\mathbf 0$, $y=0$) to $\bar n_O$; draw a fresh secret permutation $\tau$ of $[\bar n_O]$
\State \Bdo $(B_R,B_O,B_y)\gets\textsc{ChooseBounds}$ from public reference data or domain knowledge \Comment{$B_R$ covers $\Rp$'s intercept}
\State $(\Delta,\Delta_{\neg yy})\gets\textsc{DeltaRep}(B_R,B_O,B_y)$;\quad $\Delta_G\gets\textsc{DeltaRep}(B_R,B_O,0)$;\quad $\Delta_y\gets B_y^2$
\State $\textit{mode}\gets\textsf{joint}$ if $\Delta-\Delta_{\neg yy}\le10^{-9}\Delta$, otherwise $\textsf{separate}$ \Comment{release $\by^\top\by$ jointly only when it is free}
\State \Odo $(\Omega,q_0,\{\sigma_{c,s},\rho_s,\delta_s\}_s)\gets\textsc{Calibrate}(\varepsilon_{\mathrm{run}},\delta_{\mathrm{run}},f_1,f_y,\{\Delta_s\},N,\kappa)$
\State \Odo \textbf{if} $\mathrm{Ledger}[\text{base register}]+(\rho_{\mathrm{run}},\delta_{\mathrm{run}})$ exceeds the cap \textbf{then} refuse
\State \Bdo threshold key generation with a common random $a$: $s=s_R+s_O$, $\pk=(p_0,a)$ with $p_0=-as+e$, and rotation keys for $\{2^i\}$
\Statex \hspace{\algorithmicindent}(no relinearization or conjugation key)
\State \Odo fresh OPRF key $k$; PSI preprocessing on the items $\{F_k(\mathrm{id}_j)\}$ with labels $\tau(j)$ encrypted under keys derived from $F'_k(\mathrm{id}_j)$; bin sizes and partition counts public; log the run against the OPRF quota of (researcher, project)
\end{algorithmic}
\end{algorithm}

% --------------------------------------------------------------
\begin{algorithm}[p]
\caption{\textsc{Link}: labeled PSI with an OPRF (Phase~1)}
\label{alg:link}
\small
\begin{algorithmic}[1]
\State \Rdo pad the cohort identifiers with random dummies to exactly $n_R^{\max}$
\State \Rdo run the OPRF with $\Op$ on each padded identifier $\to F_k(\mathrm{id}),F'_k(\mathrm{id})$ \Comment{$\Op$ evaluates exactly $n_R^{\max}$ calls}
\State \Rdo cuckoo-hash the items $F_k(\mathrm{id})$ into the bins; send BFV encryptions of their windowed powers under $\Rp$'s own BFV key
\State \Odo evaluate the membership and label polynomials of every bin and partition; mask; $\textsc{SanitizeBFV}$; reply
\Statex \hspace{\algorithmicindent}\tocheck{The v3.1 specification only modulus-switches the PSI replies. The proof review says disclosure G4 needs circuit privacy of these replies (noise flooding or re-randomization). Confirm that SanitizeBFV is part of the protocol.}
\State \Rdo decrypt $\to$ the match set $I$ and positions $\pi(i)=\tau(j)$ for $i\in I$ (labels opened with keys from $F'_k(\mathrm{id}_i)$)
\State \Rdo $\bA\gets\sum_{i\in I}\bxd_i\bxd_i^\top$, exact and local \Comment{never transmitted, never noised}
\end{algorithmic}
\end{algorithm}

% --------------------------------------------------------------
\begin{algorithm}[p]
\caption{\textsc{Upload} (Phase~2)}
\label{alg:upload}
\small
\begin{algorithmic}[1]
\For{$t=1,\dots,\bar n_O$}
  \State \Rdo $b_t\gets\mathbb{1}[t\in\pi(I)]$;\quad $\bxd_t\gets\bigl(1,\ \mathrm{clip}(\bx_{R,i};\sqrt{B_R^2-1})\bigr)$ if $t=\pi(i)$, otherwise $\mathbf 0$
\EndFor
\State $m\gets\lceil\bar n_O/N_s\rceil$ \Comment{chunks of at most $N_s=N/2$ slots}
\For{each vector $\mathbf v\in\{\bb,\bxd_{\cdot0},\dots,\bxd_{\cdot d_R}\}$ and each chunk $h\le m$}
  \State \Rdo draw a seed $\varsigma$; $a\gets\mathrm{PRG}(\varsigma)$; $c_0\gets-a\,s_R+e+\Ecd_{\Omega_U}(\mathbf v[h])$, with $e$ from the RLWE error distribution; send $(c_0,\varsigma)$
  \State \Odo $a\gets\mathrm{PRG}(\varsigma)$;\ $c_0\gets c_0-a\,s_O$ \Comment{$(c_0,a)$ now encrypts $\mathbf v[h]$ under $s=s_R+s_O$}
\EndFor
\Statex $\Rp$ encodes at $\Omega_U=2^{68}$ (Section~\ref{sec:privacy:calib})
\end{algorithmic}
\end{algorithm}

% --------------------------------------------------------------
\begin{algorithm}[p]
\caption{\textsc{Aggregate} and pack ($\Op$, Phase~3)}
\label{alg:aggregate}
\small
\begin{algorithmic}[1]
\Require entry map $e\mapsto(U_e,V_e)$, with $\Rp$'s ciphertext vector $U_e$ and $\Op$'s plaintext vector $V_e$ in $\tau$-order:
\Statex \quad $\bB_{jk}$: $(\Enc{\bb},\ \bx_{O,j}\odot\bx_{O,k})$;\quad $\bC_{jk}$: $(\Enc{\bxd_j},\ \bx_{O,k})$;\quad $\bc_{R,j}$: $(\Enc{\bxd_j},\ \by)$;
\Statex \quad $\bc_{O,k}$: $(\Enc{\bb},\ \bx_{O,k}\odot\by)$;\quad $\by^\top\by$: $(\Enc{\bb},\ \by\odot\by)$
\For{each $e\in\mathcal E$}
  \State $T_e\gets\textsc{EvalSum}\bigl(\textsc{Rescale}\bigl(\sum_{h\le m}U_e[h]\odot\Ecd_{\Omega_V}(V_e[h])\bigr)\bigr)$ \Comment{sum the chunks first; one \textsc{EvalSum} per entry}
  \State $P_e\gets\textsc{Rescale}\bigl(T_e\odot\Ecd_{\Omega_u}(\mathbf u_{\mathrm{slot}(e)})\bigr)$ \Comment{$\mathbf u_k$: one-hot slot $k$; entry $e$ sits alone in its slot}
\EndFor
\State $M_1\gets\sum_{e\in\mathcal E_G}P_e$;\quad $M_2\gets\sum_{e\in\mathcal E_2}P_e$;\quad $M_y\gets P_{\by^\top\by}$ if $\textit{mode}=\textsf{separate}$ \Comment{depth 2; $M_1$ depends only on $\triu\bB$ and $\bC$}
\Statex Scales: $\Rp$'s vectors at $\Omega_U=2^{68}$, the masks at $\Omega_u=q_1$, $\Op$'s data at $\Omega_V$; the final scale $\Omega_f=\Omega_U\Omega_V\Omega_u/(q_1q_2)$ is tracked exactly, and $\Rp$ decodes with it
\end{algorithmic}
\end{algorithm}

% --------------------------------------------------------------
\begin{algorithm}[p]
\caption{\textsc{Release}$(M,\sigma_c,\rho,\delta)$: one DP release (Phase~4)}
\label{alg:release}
\small
\begin{algorithmic}[1]
\State \Odo sample fresh $u$ from the key distribution and $e_0,e_1$ from the error distribution
\State \Odo $c_0'\gets c_0+p_0u+e_0$;\quad $c_1'\gets c_1+au+e_1$ \Comment{re-randomize $M=(c_0,c_1)$ under the joint $\pk$}
\State \Odo $\eta\gets(\textsc{SampleDG}(\sigma_c))^{N}$, fresh \Comment{exact discrete Gaussian on $\Z^N$}
\State \Odo $h_O\gets c_0'+c_1's_O+\eta\bmod q_0$
\State \Odo $\mathrm{Ledger}\mathrel{+}=(\rho,\delta)$; send $(c_1',h_O)$; erase $c_0',u,e_0,e_1,\eta$ \Comment{$c_0'$ is never sent}
\State \Rdo $v\gets h_O+c_1's_R\bmod q_0$;\quad \Return $\Dcd(v)$ restricted to the slots of this release \Comment{decode with the exact final scale}
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[p]
\caption{\textsc{Solve} and \textsc{Infer} ($\Rp$, Phase~5)}
\label{alg:solve}
\small
\begin{algorithmic}[1]
\Require $\bA$, $n$, $p$, slot-domain s.d.\ $\sigma_1,\sigma_2$ (and $\sigma_y$), $\vartheta^\star$, $\alpha=\alpha_{\mathrm{CI}}+\alpha_L+\alpha_U$ (default $\alpha_L=\alpha_U=\alpha/20$), $B_\beta^{\mathrm{pub}}$, $\kappa_\Psi$ (default 2), bootstrap size $B$
\State $(\hat\bB_1,\hat\bC_1)\gets\textsc{Release}(M_1,\sigma_{c,1},\rho_1,\delta_1)$;\quad $\hat\bG_1\gets\begin{psmallmatrix}\bA&\hat\bC_1\\ \hat\bC_1^\top&\hat\bB_1\end{psmallmatrix}$
\State $r_1\gets\sigma_1\bigl(2\sqrt p+2\sqrt{\ln(1/\alpha_L)}\bigr)$ \Comment{$\Pr[\lambda_{\max}(\hat\bG_1-\bG)>r_1]\le\alpha_L$ (one-sided)}
\State $\hat L\gets\max\{0,\ \lambda_{\min}(\hat\bG_1)-r_1\}$;\quad $g^\star\gets2\vartheta^\star\sigma_2\sqrt p$
\State $\lambda_I\gets\max\{0,\ g^\star+r_1-\lambda_{\min}(\hat\bG_1)\}$;\quad $\lambda_O\gets\min\{\lambda\ge0:\lambda_{\min}(\hat\bG_1+\lambda\PiO)\ge g^\star+r_1\}$ ($\infty$ if none)
\Statex \hspace{\algorithmicindent}\tocheck{Replace the bisection for lambda-O by the closed form of the proofs (Lemma S3(c)).}
\State $(\lambda,\Psi)\gets(\lambda_O,\PiO)$ if $\lambda_O\le\kappa_\Psi\lambda_I$, otherwise $(\lambda_I,\bI)$ \Comment{uses stage 1 and $\bA$ only}
\State $(\hat\bB_2,\hat\bC_2,\hat\bc_R,\hat\bc_O,\widehat{yy})\gets\textsc{Release}(M_2,\sigma_{c,2},\rho_2,\delta_2)$; if separate, $\widehat{yy}\gets\textsc{Release}(M_y,\sigma_{c,y},\rho_y,\delta_y)$
\State assemble $\hat\bG_2$, $\hat\bc_2$;\quad $\hat\vartheta\gets\lambda_{\min}(\hat\bG_2+\lambda\Psi)/(2\sigma_2\sqrt p)$;\quad \textbf{if} $\hat\vartheta<1$ \textbf{then return} \textsf{refused} \Comment{budget is spent}
\State $\tilde\bP\gets(\hat\bG_2+\lambda\Psi)^{-1}$;\quad $\tilde\btheta\gets\tilde\bP\hat\bc_2$ \Comment{the bias correction of Section~\ref{sec:bias} is optional, off by default}
\State $\widetilde{\mathrm{RSS}}\gets\widehat{yy}-\hat\bc_2^\top\tilde\btheta-\lambda\,\tilde\btheta^\top\Psi\tilde\btheta$;\quad $\sigma_{yy}\gets\sigma_2$ (joint) or $\sigma_y$ (separate)
\For{$b=1,\dots,B$} \Comment{parametric bootstrap of the mechanism: post-processing}
  \State draw $\mathbf E^*$ (symmetric, s.d.\ $\sigma_2$ on the $\Op$-dependent entries, zero on $\bA$), $\mathbf f^*\sim\mathcal N(0,\sigma_2^2\bI)$, $g^*\sim\mathcal N(0,\sigma_{yy}^2)$
  \State \textbf{keep} the draw only if $\|\tilde\bP^{1/2}\mathbf E^*\tilde\bP^{1/2}\|_{\mathrm{op}}\le\frac12$ \Comment{truncation: otherwise the moments do not exist}
  \State $\btheta^*_b\gets(\hat\bG_2+\mathbf E^*+\lambda\Psi)^{-1}(\hat\bc_2+\mathbf f^*)$;\quad $\mathrm{RSS}^*_b\gets$ the RSS formula at $(\hat\bG_2+\mathbf E^*,\hat\bc_2+\mathbf f^*,\widehat{yy}+g^*)$
\EndFor
\State $\bV_{\mathrm{mech}}\gets\widehat{\mathrm{Cov}}_b(\btheta^*_b)$;\quad $\mathrm{Corr}\gets\overline{\mathrm{RSS}^*}-\widetilde{\mathrm{RSS}}$;\quad $\tilde s^2\gets\max\{\widetilde{\mathrm{RSS}}-\mathrm{Corr},0\}/(n-p)$
\State \textbf{CI for the protocol estimand} $\btheta_\lambda=(\bG+\lambda\Psi)^{-1}\bc$:\quad $\tilde\theta_j\pm z_{1-\alpha_{\mathrm{CI}}/2}\sqrt{(\bV_{\mathrm{mech}})_{jj}}$
\State $\bV\gets\bV_{\mathrm{mech}}+\tilde s^2\,\tilde\bP\hat\bG_2\tilde\bP$ \Comment{adds sampling variance under the homoskedastic working model}
\State $U_{yy}\gets\min\bigl\{\widehat{yy}+z_{1-\alpha_U/2}\,\sigma_{yy}+B_y^2\sqrt{n\ln(2/\alpha_U)/2},\ nB_y^2\bigr\}$;\quad $\hat B_\beta\gets\min\{B_\beta^{\mathrm{pub}},\sqrt{U_{yy}/\hat L}\}$ \Comment{$\infty$ if $\hat L=0$}
\Statex \hspace{\algorithmicindent}\tocheck{U-yy as corrected by the proof review (the specification spent alpha-U twice and left sigma-yy undefined). Confirm.}
\State $\bar b_j\gets\lambda\,\|\Psi\tilde\bP\mathbf e_j\|_2\,\hat B_\beta$ \Comment{worst-case ridge bias of coordinate $j$ (first order)}
\State \textbf{CI for} $\beta_j$ (bias-aware): $\tilde\theta_j\pm\mathrm{cv}_{\alpha_{\mathrm{CI}}}\bigl(\bar b_j/\sqrt{V_{jj}}\bigr)\sqrt{V_{jj}}$, where $\mathrm{cv}_\alpha(t)$ is the $(1-\alpha)$ quantile of $|Z+t|$, $Z\sim\mathcal N(0,1)$
\State \Return $\tilde\btheta$, both intervals, $(\lambda,\Psi)$, the shrinkage $-\lambda\tilde\bP\Psi\tilde\btheta$, $\hat\vartheta$, $\sigma_2\tr(\tilde\bP)$, and the $(\varepsilon,\delta)$ spent
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[p]
\caption{Subroutines}
\label{alg:sub}
\small
\begin{algorithmic}[1]
\Function{DeltaRep}{$B_R,B_O,B_y$} \Comment{replace-one of $(\bx_O,y)$, $\bxd$ fixed; $r,o,w=B_R^2,B_O^2,B_y^2$; Proposition~\ref{prop:sens}}
  \State \textbf{if} $|r-w|\le2o$: $\Delta^2=(2o+r+w)^2/2+2rw$;\quad \textbf{else if} $r-w\ge2o$: $\Delta^2=4r(o+w)$
  \State \textbf{else}: exact candidate enumeration (three cubics), rounded up as a safeguard. Return $\Delta$, and $\Delta_{\neg yy}$ from the same maximization without the $(y^2-y'^2)^2$ term
  \State \Comment{exact suprema for $d_O\ge2$, valid but conservative for $d_O=1$; $\Delta_G=(2o+r)/\sqrt2$ if $r\le2o$, otherwise $2\sqrt{ro}$}
\EndFunction
\Statex
\Function{Calibrate}{$\varepsilon_{\mathrm{run}},\delta_{\mathrm{run}},f_1,f_y,\{\Delta_s\},N,\kappa$}
  \State $n_{\mathrm{rel}}\gets$ number of releases (2 or 3);\quad $\gamma\gets n_{\mathrm{rel}}\bigl(\delta_B+2^{-\kappa}(1+2^{-80})+2^{-97}\bigr)$;\quad $\delta_{\mathrm{DP}}\gets\delta_{\mathrm{run}}-(1+e^{\varepsilon_{\mathrm{run}}})\gamma$
  \State $\rho_{\mathrm{run}}\gets\max\{\rho:\ \inf_{a>1}\frac{e^{(a-1)(a\rho-\varepsilon_{\mathrm{run}})}}{a-1}\bigl(1-\tfrac1a\bigr)^{a}\le\delta_{\mathrm{DP}}\}$ \Comment{tight zCDP conversion}
  \State $\rho_1\gets f_1\rho_{\mathrm{run}}$;\quad $\rho_y\gets f_y\rho_{\mathrm{run}}$ (0 in joint mode);\quad $\rho_2\gets\rho_{\mathrm{run}}-\rho_1-\rho_y$
  \For{$\Omega$ in increasing admissible scales}
    \State for each release $s$: $\Delta_{\mathrm{enc},s}\gets\sqrt{2/N}\,\Omega\,\Delta_s+\Delta_{x,s}+\sqrt N$;\quad $\sigma_{c,s}\gets\Delta_{\mathrm{enc},s}/\sqrt{2\rho_s}$
    \State \textbf{if} $\sigma_{c,s}\ge2^{\kappa}B_{\mathrm{tot},s}(\Omega)/\sqrt{2\pi}$ for every $s$ \textbf{then break} \Comment{worst-case, data-independent $B_{\mathrm{tot},s}$}
  \EndFor
  \State $\sigma_s\gets\sigma_{c,s}\sqrt{N/2}/\Omega$;\quad choose $q_0>2\bigl(\Omega\,S_{\max}+2^{\kappa}B_{\mathrm{tot}}+10\,\sigma_{c,\max}\bigr)$ with $S_{\max}=\bar n_O\max(r,o,w,\sqrt{ro},\dots)$ \Comment{no wraparound}
\EndFunction
\Statex
\Function{SampleDG}{$\sigma$} exact discrete Gaussian on $\Z$ by rejection from a discrete Laplace, with big-integer
\Statex \hspace{\algorithmicindent}arithmetic, a rational $\sigma^2$ and a CSPRNG; never by rounding a floating-point normal
\EndFunction
\Statex
\Function{SanitizeBFV}{$\mathit{ct}$} \Comment{statistical flooding; BFV decryption stays exact}
  \State $\mathit{ct}\gets\mathsf{ModSwitch}_0(\mathit{ct})$;\quad $\mathbf e\gets\mathcal D_{\Z^{N_B},\,2^{\kappa}B_{\mathrm{err}}}$ \Comment{$B_{\mathrm{err}}$: bound on the error of $\mathit{ct}$ at level 0}
  \State \Return $\mathit{ct}\boxplus\mathsf{BFV.Enc}_{\pk_B}(\mathbf 0)\boxplus(\mathbf e,0)$
\EndFunction
\Statex
\Function{ChooseBounds}{$D_{\mathrm{ref}}$} grid over quantile levels of $\|\bxd\|$, $\|\bx_O\|$ and $|y|$ in public reference data; minimize
\Statex \hspace{\algorithmicindent}the first-order coefficient MSE plus the squared clipping bias
\EndFunction
\end{algorithmic}
\end{algorithm}
